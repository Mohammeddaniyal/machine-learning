\hypertarget{md_docs_2private__header__explained_autotoc_md31}{}\doxysection{\texorpdfstring{2. Public vs Private Headers}{2. Public vs Private Headers}}\label{md_docs_2private__header__explained_autotoc_md31}

\begin{DoxyItemize}
\item \texorpdfstring{$\ast$}{*}\texorpdfstring{$\ast$}{*}\+Public headers ({\ttfamily include/mlfw\+\_\+error.\+h})\texorpdfstring{$\ast$}{*}\texorpdfstring{$\ast$}{*}
\begin{DoxyItemize}
\item Part of the official API of your library.
\item Distributed to users along with the compiled library ({\ttfamily .a} or {\ttfamily .so}).
\item Users include these in their code to use your library.
\end{DoxyItemize}
\item \texorpdfstring{$\ast$}{*}\texorpdfstring{$\ast$}{*}\+Private headers ({\ttfamily private/include/\+\_\+\+\_\+\+\_\+mlfw\+\_\+error.\+h})\texorpdfstring{$\ast$}{*}\texorpdfstring{$\ast$}{*}
\begin{DoxyItemize}
\item Only used by the library’s internal {\ttfamily .c} files during compilation.
\item Not distributed to users — kept internal.
\item Users never need them, because the compiled library already contains the machine code.
\end{DoxyItemize}
\end{DoxyItemize}

\hypertarget{md_docs_2private__header__explained_autotoc_md33}{}\doxysection{\texorpdfstring{3. Why Private Headers Are Not Installed}{3. Why Private Headers Are Not Installed}}\label{md_docs_2private__header__explained_autotoc_md33}
When you build a library\+:
\begin{DoxyEnumerate}
\item {\bfseries{During compilation}}
\begin{DoxyItemize}
\item Your {\ttfamily .c} files include both public and private headers.
\item The compiler uses macros, constants, and prototypes from the private headers.
\item These get baked into the object files ({\ttfamily .o}) and then into your library ({\ttfamily .so}/{\ttfamily .a}).
\end{DoxyItemize}
\item {\bfseries{When distributing the library}}
\begin{DoxyItemize}
\item You install\+:
\begin{DoxyItemize}
\item {\ttfamily libmlfw.\+so} or {\ttfamily libmlfw.\+a}
\item {\ttfamily include/mlfw\+\_\+error.\+h}
\end{DoxyItemize}
\item You do {\bfseries{not}} install\+:
\begin{DoxyItemize}
\item {\ttfamily private/include/\+\_\+\+\_\+\+\_\+mlfw\+\_\+error.\+h}
\end{DoxyItemize}
\end{DoxyItemize}
\item {\bfseries{When users link against your library}}
\begin{DoxyItemize}
\item They only include {\ttfamily mlfw\+\_\+error.\+h}.
\item They link against the precompiled binary ({\ttfamily libmlfw.\+so}).
\item They don’t need the private header, since it was only required during compilation.
\end{DoxyItemize}
\end{DoxyEnumerate}

\hypertarget{md_docs_2private__header__explained_autotoc_md38}{}\doxysection{\texorpdfstring{✅ In Short}{✅ In Short}}\label{md_docs_2private__header__explained_autotoc_md38}

\begin{DoxyItemize}
\item Private headers are {\bfseries{needed to build the library}}, but not {\bfseries{needed to use it}}.
\item At {\bfseries{compile-\/time}}, macros and declarations from private headers guide compilation.
\item At {\bfseries{runtime}}, only the compiled machine code is used, so the private headers disappear. 
\end{DoxyItemize}
